# Pinned to the versions that produced the published results, not to the
# newest that happens to resolve. A benchmark whose dependency set floats is
# a benchmark whose numbers move without anybody touching it.
#
# Python 3.14.4 is what these were built and run against.
#
# Install:  python -m venv .venv && .venv/bin/pip install -r requirements.txt
#
# To REPRODUCE the published corpus, install `requirements.lock` instead. This
# file names what we chose; that one records what was actually installed, and
# the difference includes numba, which compiles the code that decides which
# pixels carry the payload.

# Corpus construction and scoring.
numpy==2.5.3
scipy==1.18.1
pillow==12.3.0

# Adaptive embedding. `conseal` supplies HUGO, WOW, S-UNIWARD, HILL, MiPOD,
# J-UNIWARD and UERD; `jpeglib` gives it access to quantised DCT coefficients
# without going through a decode and re-encode, which would destroy the thing
# being measured.
conseal==2025.11
jpeglib==1.0.2

# Cover ingestion.
opencv-python==5.0.0.93

# Tests.
pytest==9.1.1

# The YAML parser the published metadata is checked with. It is here rather
# than in `requirements-optional.txt` on purpose: the tests that need it read
# the HuggingFace dataset card's frontmatter, and that frontmatter is what
# decides whether the dataset loads at all for HuggingFace, Kaggle and the ML
# Commons tooling. A check on a published artefact that only runs where
# somebody happens to have the parser installed is the same defect as a check
# that cannot be re-run.
#
# Measured: it was absent from both requirement files while three tests
# imported it unguarded, so CI went red on 2026-09-23 and stayed red through
# six runs, 13 errors every time, while the suite passed on every machine that
# had it. That is exactly the failure `ci.yml` warns about a few lines above
# where it installs this file.
#
# Cooldown (baseline Section 5): 6.0.3 has been installed on the machine that
# pinned it since 2026-04-23, so it is at least 159 days old and far past the
# 7 day window. The upload date itself was not read, and the installed date is
# a lower bound rather than the real age.
PyYAML==6.0.3
