# Optional, and deliberately not in `requirements.txt`. Each one is needed for

# Reference detector and the SRM feature extractor, used by the comparison
# panel and by `generators/rich_model_baseline.py`. Not on PyPI under this
# name; install from source:
#   git clone https://github.com/daniellerch/aletheia
# See docs/ for the version this was run against.

# Charts. Only `generators/*_figure*.py` and the plotting in
# `integrations/lir/make_figure.py` need it.
matplotlib==3.11.0

# The forensic likelihood ratio library, used ONLY by
# `integrations/lir/test_against_lir.py` to check our metric against an
# independent implementation. It pulls pymc, pytensor, numba, llvmlite, optuna
# and scikit-learn, roughly 2 GB, to compute a handful of scalars. That is why
# nothing in `integrations/lir/` imports it outside the tests, and why those
# tests skip when it is absent.
lir==1.8.0

# The reference validator for the Croissant record, which is the file that
# decides whether a dataset loads for Hugging Face, Kaggle and the ML Commons
# tooling. `generators/validate_croissant.py` runs a structural check without
# it and defers to it when present.
#
# It belongs here rather than in `requirements.txt` because nothing in the
# build needs it, and it belongs here rather than nowhere because it was
# nowhere: the record was validated once, by hand, on a machine where the
# package no longer exists. A check that cannot be re-run is not a check.
#
# Pinned 2026-09-21, on the machine that installed it. 1.1.0 was uploaded on
# 2026-04-16, so it was 158 days old and well past the 7 day cooldown baseline
# Section 5 asks for. It validated all three tier records the same evening.
#
# This line was unpinned on purpose before that, because writing a version
# nobody had installed would have been a guess dressed as a pin. An earlier
# draft of this file did exactly that and named a version that has never
# existed.
mlcroissant==1.1.0

# The Hansken extraction plugin SDK, used only by `integrations/hansken/`.
hansken-extraction-plugin==0.10.0
logbook==1.8.2
